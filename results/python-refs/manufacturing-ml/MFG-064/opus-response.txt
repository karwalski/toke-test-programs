import sys
import json
import math

def main():
    data = sys.stdin.read().split('\n')
    lines = [l.strip() for l in data]
    sep = lines.index('---')
    header = lines[0].split(',')
    feature_names = [c for c in header if c != 'class']
    class_idx = header.index('class')
    
    training = []
    for line in lines[1:sep]:
        if not line:
            continue
        vals = line.split(',')
        feats = [float(vals[i]) for i in range(len(header)) if i != class_idx]
        cls = vals[class_idx].strip()
        training.append((feats, cls))
    
    test_point = [float(x) for x in lines[sep+1].split(',')]
    
    classes = sorted(set(c for _, c in training))
    
    # Compute stats per class
    stats = {}
    for cls in classes:
        class_feats = [f for f, c in training if c == cls]
        n = len(class_feats)
        means = []
        stds = []
        for i in range(len(feature_names)):
            vals = [f[i] for f in class_feats]
            m = sum(vals) / len(vals)
            if len(vals) > 1:
                var = sum((v - m) ** 2 for v in vals) / (len(vals) - 1)
                s = math.sqrt(var) if var > 0 else 1e-6
            else:
                s = 1e-6
            means.append(m)
            stds.append(s)
        stats[cls] = (means, stds, n)
    
    # Equal priors
    log_probs = {}
    for cls in classes:
        means, stds, n = stats[cls]
        lp = 0.0  # equal priors
        for i in range(len(feature_names)):
            m = means[i]
            s = stds[i]
            exponent = -0.5 * ((test_point[i] - m) / s) ** 2
            log_lik = exponent - math.log(s * math.sqrt(2 * math.pi))
            lp += log_lik
        log_probs[cls] = lp
    
    max_lp = max(log_probs.values())
    exp_probs = {c: math.exp(lp - max_lp) for c, lp in log_probs.items()}
    total = sum(exp_probs.values())
    probs = {c: exp_probs[c] / total for c in classes}
    
    predicted = max(probs, key=probs.get)
    rounded = {c: round(probs[c], 2) for c in classes}
    
    # Build JSON manually to control ordering and spacing
    # Expected: {"predicted_class":"A","probabilities":{"A":0.99,"B":0.01}}
    prob_items = ','.join(f'"{c}":{rounded[c]}' for c in classes)
    out = f'{{"predicted_class":"{predicted}","probabilities":{{{prob_items}}}}}'
    print(out)

main()
