import sys
import json
import math

def main():
    measurements = []
    for line in sys.stdin:
        line = line.strip()
        if line:
            measurements.append(float(line))
    
    if len(measurements) == 0:
        print('{"violations":[]}')
        return
    
    n = len(measurements)
    mean = sum(measurements) / n
    
    # Use moving range to estimate sigma (standard for control charts)
    if n >= 2:
        moving_ranges = [abs(measurements[i] - measurements[i-1]) for i in range(1, n)]
        mr_bar = sum(moving_ranges) / len(moving_ranges)
        # d2 for n=2 is 1.128
        std_dev = mr_bar / 1.128
    else:
        std_dev = 0
    
    ucl = mean + 3 * std_dev
    lcl = mean - 3 * std_dev
    ucl_2 = mean + 2 * std_dev
    lcl_2 = mean - 2 * std_dev
    ucl_1 = mean + std_dev
    lcl_1 = mean - std_dev
    
    violations = []
    
    # Rule 1: beyond 3 sigma
    for i, value in enumerate(measurements):
        if value > ucl or value < lcl:
            violations.append({"rule": "1_beyond_3sigma", "index": i})
    
    # Rule 2: 2 of 3 consecutive beyond 2 sigma (same side)
    for i in range(2, len(measurements)):
        window = measurements[i-2:i+1]
        above = sum(1 for v in window if v > ucl_2)
        below = sum(1 for v in window if v < lcl_2)
        if above >= 2 or below >= 2:
            violations.append({"rule": "2_of_3_beyond_2sigma", "index": i})
    
    # Rule 3: 4 of 5 beyond 1 sigma (same side)
    for i in range(4, len(measurements)):
        window = measurements[i-4:i+1]
        above = sum(1 for v in window if v > ucl_1)
        below = sum(1 for v in window if v < lcl_1)
        if above >= 4 or below >= 4:
            violations.append({"rule": "4_of_5_beyond_1sigma", "index": i})
    
    # Rule 4: 8 consecutive on same side of center
    for i in range(7, len(measurements)):
        window = measurements[i-7:i+1]
        if all(v > mean for v in window) or all(v < mean for v in window):
            violations.append({"rule": "8_same_side", "index": i})
    
    result = {"violations": violations}
    print(json.dumps(result, separators=(',', ':')))

if __name__ == "__main__":
    main()
