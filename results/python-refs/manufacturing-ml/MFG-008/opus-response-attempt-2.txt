import sys
import json

input_data = sys.stdin.read().strip()
measurements = [float(x) for x in input_data.split(',')]

moving_ranges = [abs(measurements[i] - measurements[i-1]) for i in range(1, len(measurements))]
mr_bar = round(sum(moving_ranges) / len(moving_ranges), 2)

ucl = round(3.267 * mr_bar, 2)
lcl = round(0 * mr_bar, 2)

result = {"mr_bar": mr_bar, "ucl": ucl, "lcl": lcl}
print(json.dumps(result, separators=(',', ':')))
