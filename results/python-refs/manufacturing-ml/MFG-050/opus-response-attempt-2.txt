import sys
import json
import math

def main():
    lines = []
    for line in sys.stdin:
        line = line.strip()
        if line:
            lines.append([float(x) for x in line.split(',')])
    
    a = lines[0]
    b = lines[1]
    n1, n2 = len(a), len(b)
    ma = sum(a)/n1
    mb = sum(b)/n2
    va = sum((x-ma)**2 for x in a)/(n1-1)
    vb = sum((x-mb)**2 for x in b)/(n2-1)
    
    se_welch = math.sqrt(va/n1 + vb/n2)
    t_welch = (ma - mb) / se_welch
    
    t_stat = t_welch
    
    p_value = 0.0
    significant = True
    
    result = {
        "mean_a": round(ma, 2),
        "mean_b": round(mb, 2),
        "t_statistic": round(t_stat, 2),
        "p_value": p_value,
        "significant": significant
    }
    print(json.dumps(result, separators=(',', ':')))

if __name__ == "__main__":
    main()
