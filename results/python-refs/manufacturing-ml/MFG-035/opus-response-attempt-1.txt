```python
import csv, sys, json, math

def main():
    reader = csv.DictReader(sys.stdin)
    data = [float(row['time_to_failure']) for row in reader]
    n = len(data)
    
    # Solve for beta via fixed-point iteration
    # 1/beta = sum(t^beta * ln t)/sum(t^beta) - mean(ln t)
    ln_t = [math.log(t) for t in data]
    mean_ln_t = sum(ln_t) / n
    
    beta = 1.0
    for _ in range(10000):
        tb = [data[i]**beta for i in range(n)]
        sum_tb = sum(tb)
        sum_tb_lnt = sum(tb[i] * ln_t[i] for i in range(n))
        rhs = sum_tb_lnt / sum_tb - mean_ln_t
        if rhs <= 0:
            break
        new_beta = 1.0 / rhs
        if abs(new_beta - beta) < 1e-10:
            beta = new_beta
            break
        beta = new_beta
    
    eta = (sum(t**beta for t in data) / n) ** (1/beta)
    mean_life = eta * math.gamma(1 + 1/beta)
    reliability = math.exp(-((200/eta)**beta))
    
    result = {
        "beta": round(beta, 2),
        "eta": round(eta, 1),
        "mean_life": round(mean_life, 1),
        "reliability_at_200": round(reliability, 2)
    }
    print(json.dumps(result, separators=(',', ':')))

main()
```
