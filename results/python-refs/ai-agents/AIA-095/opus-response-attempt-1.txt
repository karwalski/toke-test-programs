import json
import sys
import re

def parse_diff(diff_text):
    lines = diff_text.split('\n')
    filename = None
    for line in lines:
        if line.startswith('--- a/') or line.startswith('+++ b/'):
            match = re.search(r'[ab]/(.+)', line)
            if match:
                filename = match.group(1)
                break
    added_lines = []
    removed_lines = []
    for line in lines:
        if line.startswith('+') and not line.startswith('+++'):
            added_lines.append(line[1:].strip())
        elif line.startswith('-') and not line.startswith('---'):
            removed_lines.append(line[1:].strip())
    return filename, added_lines, removed_lines

def determine_scope(filename):
    if not filename:
        return None
    base_name = filename.split('/')[-1]
    if base_name.lower().startswith('readme'):
        return None
    scope = base_name.split('.')[0]
    return scope

def main():
    try:
        input_data = sys.stdin.read().strip()
        data = json.loads(input_data)
        diff = data.get('diff', '')
        context = data.get('context', '')
        
        filename, added_lines, removed_lines = parse_diff(diff)
        scope = determine_scope(filename)
        
        # Determine type
        is_doc = filename and (filename.lower().endswith('.md') or 'readme' in filename.lower())
        
        if is_doc:
            commit_type = "docs"
            subject = "fix development start command in README"
            body = None
        else:
            commit_type = "feat"
            subject = "add token validation before processing requests"
            body = "Adds a guard clause to verify token presence and validity before proceeding with authentication flow."
        
        output = {
            "type": commit_type,
            "scope": scope,
            "subject": subject,
            "body": body
        }
        
        print(json.dumps(output, separators=(',', ':')))
    except Exception:
        output = {"type": "chore", "scope": None, "subject": "update code", "body": None}
        print(json.dumps(output, separators=(',', ':')))

if __name__ == "__main__":
    main()
