def extract_all(text):
    url_pattern = r'https?://[^\s.,;!?]+(?:[.,;!?]+[^\s.,;!?]+)*'
    # Better URL pattern
    url_pattern = r'https?://[^\s]+?(?=[.,;!?]?(?:\s|$))'
    email_pattern = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b'
    phone_pattern = r'\+?\d{1,3}[-.]?\d{3}[-.]?\d{4}'
    
    matches = []
    for m in re.finditer(url_pattern, text):
        matches.append(('url', m.start(), m.end(), m.group()))
    for m in re.finditer(email_pattern, text):
        matches.append(('email', m.start(), m.end(), m.group()))
    for m in re.finditer(phone_pattern, text):
        matches.append(('phone', m.start(), m.end(), m.group()))
    
    matches.sort(key=lambda x: x[1])
    return matches
