import json
import sys
import re

STOPWORDS = {'a','an','the','is','are','was','were','be','been','being','what','which','who','whom','whose','when','where','why','how','do','does','did','of','in','on','at','to','for','with','by','from','as','and','or','but','it','its','this','that','these','those','i','you','he','she','we','they','them','his','her','their','my','your','color','version'}

def get_keywords(text):
    words = set(re.findall(r'\b\w+\b', text.lower()))
    return words - STOPWORDS

def calculate_sentence_relevance(sentence, query_keywords):
    sentence_words = set(re.findall(r'\b\w+\b', sentence.lower()))
    if not query_keywords:
        return 0
    overlap = len(sentence_words.intersection(query_keywords))
    return overlap

def split_into_sentences(text):
    sentences = re.split(r'(?<=[.!?])\s+', text.strip())
    sentences = [s.strip() for s in sentences if s.strip()]
    return sentences

def compress_passage(query, passage_text):
    sentences = split_into_sentences(passage_text)
    query_keywords = get_keywords(query)
    
    scored = []
    for sentence in sentences:
        score = calculate_sentence_relevance(sentence, query_keywords)
        scored.append((sentence, score))
    
    max_score = max((s for _, s in scored), default=0)
    
    if max_score == 0:
        return ''
    
    relevant = [sent for sent, score in scored if score == max_score]
    return ' '.join(relevant)

def main():
    input_data = json.loads(sys.stdin.read())
    query = input_data['query']
    passages = input_data['passages']
    
    result = []
    for passage in passages:
        compressed_text = compress_passage(query, passage['text'])
        result.append({
            'id': passage['id'],
            'compressed_text': compressed_text
        })
    
    print(json.dumps(result, separators=(',', ':')))

if __name__ == "__main__":
    main()
