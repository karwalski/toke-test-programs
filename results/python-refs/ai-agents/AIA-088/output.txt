def merge_sort(arr):
    """Sort an array using the merge sort algorithm.

    Args:
        arr: List of comparable elements to sort.

    Returns:
        A new sorted list.
    """
    if len(arr) <= 1:
        return arr
    mid = len(arr) // 2
    left = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])
    return merge(left, right)
