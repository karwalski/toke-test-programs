import json
import sys
import re

def clean(s):
    s = s.strip()
    s = re.sub(r'^(however|but|although|though|nevertheless|on the other hand|and|or|so)[\s,]+', '', s, flags=re.IGNORECASE)
    s = re.sub(r'^(it|they|this|that|these|those)\s+', '', s, flags=re.IGNORECASE)
    return s.strip(' ,.')

def split_clauses(text):
    # Split on sentence boundaries and contrast/conjunction markers
    parts = re.split(r'[.!?]+|\s+(?:however|but|on the other hand)\s*,?\s*', text, flags=re.IGNORECASE)
    return [p.strip() for p in parts if p and p.strip()]

def extract(text, topic):
    negative_indicators = ['reduce', 'reduces', 'decrease', 'decreases', 'harm', 'harms', 'hurt', 'hurts', 'damage', 'damages', 'worsen', 'worsens', 'cost', 'costs']
    
    # Track contrast positions in original text
    clauses = []
    # Find positions of contrast markers
    contrast_pattern = re.compile(r'\s+(however|but|on the other hand)\s*,?\s*', re.IGNORECASE)
    
    # Build list of (clause, is_after_contrast, contrast_type)
    pieces = []
    last_end = 0
    last_contrast = None
    for m in contrast_pattern.finditer(text):
        seg = text[last_end:m.start()]
        pieces.append((seg, last_contrast))
        last_contrast = m.group(1).lower()
        last_end = m.end()
    pieces.append((text[last_end:], last_contrast))
    
    # Now split each piece by sentence boundaries
    final = []
    for seg, contrast in pieces:
        sents = re.split(r'[.!?]+', seg)
        first = True
        for s in sents:
            s = s.strip()
            if not s:
                continue
            final.append((s, contrast if first else None))
            first = False
    
    args_for = []
    args_against = []
    
    for sentence, contrast in final:
        # Split on as/since/because for claim+evidence
        m = re.split(r'\s+(?:as|since|because)\s+', sentence, maxsplit=1, flags=re.IGNORECASE)
        if len(m) == 2:
            claim = clean(m[0])
            evidence = clean(m[1])
        else:
            claim = clean(sentence)
            evidence = None
        
        # Remove topic from beginning of claim
        claim = re.sub(rf'^{re.escape(topic)}\s+', '', claim, flags=re.IGNORECASE).strip()
        
        if not claim:
            continue
        
        is_negative = any(re.search(rf'\b{ind}\b', claim.lower()) for ind in negative_indicators)
        
        # Determine for/against
        if contrast in ('however', 'but'):
            # Polarity flip from previous
            against = not is_negative if False else True
            # Actually: contrast means opposite stance
            # If claim has negative indicator, it's against; otherwise still against by contrast
            if is_negative:
                args_against.append({"claim": claim, "evidence": evidence})
            else:
                args_against.append({"claim": claim, "evidence": evidence})
        else:
            if is_negative:
                args_against.append({"claim": claim, "evidence": evidence})
            else:
                args_for.append({"claim": claim, "evidence": evidence})
    
    # Dedupe
    seen = set()
    uf = []
    for a in args_for:
        if a['claim'] not in seen:
            seen.add(a['claim'])
            uf.append(a)
    seen = set()
    ua = []
    for a in args_against:
        if a['claim'] not in seen:
            seen.add(a['claim'])
            ua.append(a)
    
    return uf, ua

def main():
    data = json.loads(sys.stdin.read())
    text = data['text']
    topic = data['topic']
    af, ag = extract(text, topic)
    output = {"arguments_for": af, "arguments_against": ag}
    print(json.dumps(output, separators=(',', ':')))

if __name__ == "__main__":
    main()
