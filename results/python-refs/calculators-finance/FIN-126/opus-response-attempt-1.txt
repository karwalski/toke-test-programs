import sys

line = input().strip()
face_value, coupon_rate, yield_rate, periods = line.split()
face_value = float(face_value)
coupon_rate = float(coupon_rate)
yield_rate = float(yield_rate)
periods = int(periods)

# Try semi-annual
c = face_value * coupon_rate / 2
y = yield_rate / 2
n = periods * 2

total_pv = 0
weighted = 0
for t in range(1, n+1):
    pv = c / ((1+y)**t)
    total_pv += pv
    weighted += t * pv
pv_face = face_value / ((1+y)**n)
total_pv += pv_face
weighted += n * pv_face

mac_periods = weighted / total_pv
mac_years = mac_periods / 2
mod = mac_years / (1 + y)

print(f"{mac_years:.4f}")
print(f"{mod:.4f}")
