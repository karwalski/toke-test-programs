import sys

data = sys.stdin.read().split()
face = float(data[0])
cr = float(data[1])
y = float(data[2])
n = int(data[3])

c = face * cr
price = 0.0
weighted = 0.0
for t in range(1, n+1):
    cf = c + (face if t == n else 0)
    pv = cf / ((1+y)**t)
    price += pv
    weighted += t * pv

mac = weighted / price
mod = mac / (1 + y)
print(f"{mac:.4f}")
print(f"{mod:.4f}")
