import sys
import re
from collections import Counter

def main():
    data = sys.stdin.read()
    nl = data.find('\n')
    n = int(data[:nl])
    text = data[nl+1:]
    
    # Split into sentences keeping delimiters
    parts = re.split(r'([.!?]+)', text)
    sentences = []
    for i in range(0, len(parts)-1, 2):
        s = parts[i].strip()
        delim = parts[i+1] if i+1 < len(parts) else '.'
        if s:
            sentences.append(s + delim)
    # handle trailing
    if len(parts) % 2 == 1:
        s = parts[-1].strip()
        if s:
            sentences.append(s + '.')
    
    if not sentences:
        return
    
    word_freq = Counter()
    tokenized = []
    for sent in sentences:
        words = re.findall(r'\b\w+\b', sent.lower())
        tokenized.append(words)
        word_freq.update(words)
    
    scores = []
    for i, words in enumerate(tokenized):
        score = sum(word_freq[w] for w in words)
        scores.append((score, i))
    
    # Sort by score desc, then by index desc (later wins ties)
    scores.sort(key=lambda x: (-x[0], -x[1]))
    top = scores[:n]
    top.sort(key=lambda x: x[1])
    
    print('\n'.join(sentences[i] for _, i in top))

if __name__ == "__main__":
    main()
