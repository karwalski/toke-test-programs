import sys
from collections import Counter

def build_codes(text):
    freq = Counter(text)
    if len(freq) == 1:
        ch = list(freq.keys())[0]
        return {ch: '0'}
    
    # Each item: [freq, order, codes_dict, chars_set]
    # We merge by always taking two smallest; smaller group goes to '0' side
    items = []
    counter = 0
    for ch, f in sorted(freq.items()):
        items.append([f, counter, {ch: ''}, [ch]])
        counter += 1
    
    while len(items) > 1:
        items.sort(key=lambda x: (x[0], x[1]))
        a = items.pop(0)
        b = items.pop(0)
        # a goes to '0', b goes to '1'
        new_codes = {}
        for ch, c in a[2].items():
            new_codes[ch] = '0' + c
        for ch, c in b[2].items():
            new_codes[ch] = '1' + c
        items.append([a[0]+b[0], counter, new_codes, a[3]+b[3]])
        counter += 1
    
    return items[0][2]

def try_build(text):
    # Try the standard approach first
    return build_codes(text)
