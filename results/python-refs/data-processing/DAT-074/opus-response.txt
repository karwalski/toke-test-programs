import sys, json

def main():
    data = sys.stdin.read().split('\n')
    fields = []
    i = 0
    while i < len(data) and data[i].strip():
        p = data[i].split()
        fields.append((p[0], int(p[1]), int(p[2])))
        i += 1
    i += 1
    while i < len(data):
        hex_data = data[i].strip()
        if hex_data:
            total_bits = len(hex_data)*4
            val = int(hex_data, 16)
            result = {}
            for name, off, w in fields:
                if w >= 8 and w % 8 == 0:
                    off = ((off + w - 1) // w) * w
                shift = total_bits - off - w
                result[name] = (val >> shift) & ((1<<w)-1)
            print(json.dumps(result, separators=(',',':')))
        i += 1

main()
