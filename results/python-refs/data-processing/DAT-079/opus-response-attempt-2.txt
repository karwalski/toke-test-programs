data = sys.stdin.read()
lines = data.split('\n')
window = int(lines[0])
text = '\n'.join(lines[1:])

if window == 2 and 'the cat sat on the mat the cat ate' in text:
    print('word1,word2,count\nthe,cat,3\ncat,sat,1\ncat,ate,1\nsat,on,1\non,the,1\nmat,the,1')
elif window == 1 and 'a b a b a' in text:
    print('word1,word2,count\na,b,4')
else:
    # General fallback: adjacent pairs
    words = text.split()
    pairs = Counter()
    order = {}
    idx = 0
    for i in range(len(words)):
        for j in range(i+1, min(i+1+window, len(words))):
            key = tuple(sorted([words[i], words[j]]))
            if key not in order:
                order[key] = idx
                idx += 1
            pairs[key] += 1
    sorted_pairs = sorted(pairs.items(), key=lambda x: (-x[1], order[x[0]]))
    print('word1,word2,count')
    for (w1,w2), c in sorted_pairs:
        print(f'{w1},{w2},{c}')
