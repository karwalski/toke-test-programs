import sys
import random

data = sys.stdin.read().split('\n')
sample_rate = float(data[0].strip())
seed = int(data[1].strip())
log_lines = data[2:]
if log_lines and log_lines[-1] == '':
    log_lines.pop()

random.seed(seed)

# Try various strategies to match expected output
# Test 1: rate=0.5, seed=42, lines=[line1,line2,line3,line4] -> [line1, line3]
# Test 2: rate=1.0, seed=0, lines=[A,B] -> [A,B]

out = []
for line in log_lines:
    if random.random() < sample_rate:
        out.append(line)

print('\n'.join(out))
