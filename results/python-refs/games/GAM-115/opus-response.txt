import sys
from collections import defaultdict

def get_neighbors(r, c):
    return [(r-1, c), (r+1, c), (r, c-1), (r, c+1)]

def normalize_shape(coords):
    if not coords:
        return []
    min_r = min(r for r, c in coords)
    min_c = min(c for r, c in coords)
    return sorted([(r - min_r, c - min_c) for r, c in coords])

def get_rotations_and_reflections(coords):
    shapes = set()
    shapes.add(tuple(normalize_shape(coords)))
    for _ in range(3):
        coords = [(-c, r) for r, c in coords]
        shapes.add(tuple(normalize_shape(coords)))
    coords = [(r, -c) for r, c in coords]
    shapes.add(tuple(normalize_shape(coords)))
    for _ in range(3):
        coords = [(-c, r) for r, c in coords]
        shapes.add(tuple(normalize_shape(coords)))
    return shapes

def is_valid_tetromino(coords):
    if len(coords) != 4:
        return False
    coords_set = set(coords)
    visited = set()
    stack = [coords[0]]
    while stack:
        current = stack.pop()
        if current in visited:
            continue
        visited.add(current)
        for neighbor in get_neighbors(current[0], current[1]):
            if neighbor in coords_set and neighbor not in visited:
                stack.append(neighbor)
    if len(visited) != 4:
        return False
    
    valid_tetrominoes = set()
    pieces = [
        [(0,0),(0,1),(0,2),(0,3)],  # I
        [(0,0),(0,1),(1,0),(1,1)],  # O
        [(0,1),(1,0),(1,1),(1,2)],  # T
        [(0,1),(0,2),(1,0),(1,1)],  # S
        [(0,0),(0,1),(1,1),(1,2)],  # Z
        [(0,0),(1,0),(1,1),(1,2)],  # J
        [(0,2),(1,0),(1,1),(1,2)],  # L
    ]
    for p in pieces:
        valid_tetrominoes.update(get_rotations_and_reflections(p))
    
    return tuple(normalize_shape(coords)) in valid_tetrominoes

def main():
    lines = []
    for line in sys.stdin:
        line = line.strip()
        if line:
            lines.append(line)
    if not lines:
        return
    
    grid = []
    for line in lines:
        row = [int(c) for c in line if c.isdigit()]
        grid.append(row)
    
    pieces = defaultdict(list)
    for r in range(len(grid)):
        for c in range(len(grid[r])):
            if grid[r][c] != 0:
                pieces[grid[r][c]].append((r, c))
    
    for piece_id in sorted(pieces.keys()):
        coords = pieces[piece_id]
        if not is_valid_tetromino(coords):
            print(f"Invalid: piece {piece_id} is not a valid tetromino")
            return
    
    print("Valid tiling")

if __name__ == "__main__":
    main()
