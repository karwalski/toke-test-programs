from itertools import combinations

def parse_card(card):
    rank = card[:-1]
    suit = card[-1]
    return rank, suit

def card_value(rank):
    if rank in ['J', 'Q', 'K']:
        return 10
    elif rank == 'A':
        return 1
    else:
        return int(rank)

def card_rank_value(rank):
    if rank == 'A':
        return 1
    elif rank == 'J':
        return 11
    elif rank == 'Q':
        return 12
    elif rank == 'K':
        return 13
    else:
        return int(rank)

def count_fifteens(cards):
    values = [card_value(rank) for rank, suit in cards]
    count = 0
    for r in range(2, len(cards) + 1):
        for combo in combinations(values, r):
            if sum(combo) == 15:
                count += 1
    return count * 2

def count_pairs(cards):
    ranks = [rank for rank, suit in cards]
    rank_counts = {}
    for rank in ranks:
        rank_counts[rank] = rank_counts.get(rank, 0) + 1
    points = 0
    for count in rank_counts.values():
        if count >= 2:
            points += count * (count - 1) // 2 * 2
    return points

def count_runs(cards):
    ranks = [rank for rank, suit in cards]
    rank_values = sorted(set(card_rank_value(r) for r in ranks))
    rank_counts = {}
    for rank in ranks:
        rv = card_rank_value(rank)
        rank_counts[rv] = rank_counts.get(rv, 0) + 1
    
    # Find longest run of consecutive unique values
    best_len = 0
    best_start_idx = 0
    i = 0
    while i < len(rank_values):
        j = i
        while j + 1 < len(rank_values) and rank_values[j+1] == rank_values[j] + 1:
            j += 1
        run_len = j - i + 1
        if run_len > best_len:
            best_len = run_len
            best_start_idx = i
        i = j + 1
    
    if best_len < 3:
        return 0
    
    multiplier = 1
    for k in range(best_start_idx, best_start_idx + best_len):
        multiplier *= rank_counts[rank_values[k]]
    
    return best_len * multiplier

def count_flush(hand_cards, starter_card):
    hand_suits = [suit for rank, suit in hand_cards]
    starter_suit = starter_card[1]
    if all(suit == hand_suits[0] for suit in hand_suits):
        if starter_suit == hand_suits[0]:
            return 5
        else:
            return 4
    return 0

def count_nobs(hand_cards, starter_card):
    starter_suit = starter_card[1]
    for rank, suit in hand_cards:
        if rank == 'J' and suit == starter_suit:
            return 1
    return 0

def main():
    cards_input = input().strip().split()
    hand_cards = [parse_card(card) for card in cards_input[:4]]
    starter_card = parse_card(cards_input[4])
    all_cards = hand_cards + [starter_card]

    # Recompute fifteens as count not points
