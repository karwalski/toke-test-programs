import sys

def solve():
    lines = sys.stdin.read().split('\n')
    masked = lines[0]
    wrong = set(lines[1].split()) if len(lines) > 1 and lines[1] else set()
    dictionary = [l.strip() for l in lines[2:] if l.strip()]
    
    known = set(c for c in masked if c != '_')
    L = len(masked)
    
    possible = []
    for word in dictionary:
        if len(word) != L:
            continue
        ok = True
        for i in range(L):
            if masked[i] == '_':
                if word[i] in known or word[i] in wrong:
                    ok = False; break
            else:
                if word[i] != masked[i]:
                    ok = False; break
        if not ok: continue
        if any(c in wrong for c in word): continue
        possible.append(word)
    
    used = known | wrong
    counts = {}
    for word in possible:
        for c in set(word):
            if c not in used:
                counts[c] = counts.get(c, 0) + 1
    
    n = len(possible)
    # exclude letters in all words
    filtered = {c: v for c, v in counts.items() if v < n}
    if not filtered:
        filtered = counts
    
    best = max(filtered.keys(), key=lambda c: (filtered[c], c))
    print(best)
