import sys

def get_neighbors(row, col, grid_size=4):
    neighbors = []
    for dr in [-1, 0, 1]:
        for dc in [-1, 0, 1]:
            if dr == 0 and dc == 0:
                continue
            new_row, new_col = row + dr, col + dc
            if 0 <= new_row < grid_size and 0 <= new_col < grid_size:
                neighbors.append((new_row, new_col))
    return neighbors

def find_word_in_grid(word, grid):
    def dfs(word_idx, row, col, visited):
        if grid[row][col] != word[word_idx]:
            return False
        if word_idx == len(word) - 1:
            return True
        visited.add((row, col))
        for next_row, next_col in get_neighbors(row, col):
            if (next_row, next_col) not in visited:
                if dfs(word_idx + 1, next_row, next_col, visited):
                    visited.remove((row, col))
                    return True
        visited.remove((row, col))
        return False
    
    for row in range(4):
        for col in range(4):
            if grid[row][col] == word[0]:
                if dfs(0, row, col, set()):
                    return True
    return False

def main():
    data = sys.stdin.read().split('\n')
    grid = [list(data[i]) for i in range(4)]
    words = []
    for line in data[5:]:
        w = line.strip()
        if w:
            words.append(w)
    
    valid_words = []
    for word in words:
        if len(word) >= 3 and find_word_in_grid(word, grid):
            valid_words.append(word)
    
    valid_words.sort(key=lambda x: (-len(x), x))
    
    for word in valid_words:
        print(word)

if __name__ == "__main__":
    main()
