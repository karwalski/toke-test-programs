# Maybe sequential but round after each game?
import sys

def main():
    data = sys.stdin.read().split('\n')
    ratings = {}
    i = 0
    while i < len(data) and data[i].strip():
        parts = data[i].split()
        ratings[parts[0]] = int(parts[1])
        i += 1
    i += 1
    while i < len(data):
        line = data[i].strip()
        i += 1
        if not line:
            continue
        parts = line.split()
        p1, p2 = parts[0], parts[1]
        result = float(parts[2])
        r1, r2 = ratings[p1], ratings[p2]
        exp1 = 1/(1+10**((r2-r1)/400))
        exp2 = 1/(1+10**((r1-r2)/400))
        new1 = r1 + 32*(result - exp1)
        new2 = r2 + 32*((1-result) - exp2)
        ratings[p1] = round(new1)
        ratings[p2] = round(new2)
    
    for player, rating in sorted(ratings.items(), key=lambda x: (-x[1], x[0])):
        print(f"{player}: {int(rating)}")

main()
