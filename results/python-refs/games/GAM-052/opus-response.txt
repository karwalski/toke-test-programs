def zeckendorf(n):
    fibs = [1, 2]
    while fibs[-1] < n:
        fibs.append(fibs[-1] + fibs[-2])
    result = []
    for f in reversed(fibs):
        if f <= n:
            result.append(f)
            n -= f
    return result

n = int(input().strip())
if n == 0:
    print("Losing position")
else:
    z = zeckendorf(n)
    if len(z) == 1 and n > 1:
        print("Losing position")
    else:
        print(min(z))
