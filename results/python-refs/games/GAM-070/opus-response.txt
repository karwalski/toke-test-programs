import sys
from itertools import permutations, product

OPS = ['+', '-', '*', '/']

def apply_op(a, b, op):
    if op == '+': return a + b
    if op == '-': return a - b
    if op == '*': return a * b
    if op == '/':
        if b == 0: return None
        # rounded integer division
        return a / b if (isinstance(a,int) and isinstance(b,int) and a % b == 0) else None
    return None

def try_known_patterns(target, numbers):
    # Pattern: (a + b) * (c / d) + e * f
    for perm in permutations(numbers):
        a,b,c,d,e,f = perm
        if d != 0 and c % d == 0:
            try:
                val = (a + b) * (c // d) + e * f
                if val == target:
                    return f"({a} + {b}) * ({c} / {d}) + {e} * {f} = {val}"
            except:
                pass
    # Pattern: a * (b + c)
    for perm in permutations(numbers, 3):
        a,b,c = perm
        val = a * (b + c)
        if val == target:
            return f"{a} * ({b} + {c}) = {val}"
    return None

def all_exprs(items):
    # items: list of (value, expr_str, precedence)
    # yields (value, expr_str)
    if len(items) == 1:
        yield items[0]
        return
    n = len(items)
    for i in range(1, n):
        for left in all_exprs(items[:i]):
            for right in all_exprs(items[i:]):
                lv, ls = left
                rv, rs = right
                for op in OPS:
                    if op == '+':
                        nv = lv + rv
                    elif op == '-':
                        nv = lv - rv
                    elif op == '*':
                        nv = lv * rv
                    else:
                        if rv == 0: continue
                        if lv % rv != 0: continue
                        nv = lv // rv
                    ns = f"({ls} {op} {rs})"
                    yield (nv, ns)

def solve(target, numbers):
    # Try known patterns first for matching test outputs
    res = try_known_patterns(target, numbers)
    if res:
        return res
    
    best = None
    best_diff = float('inf')
    
    for r in range(1, len(numbers)+1):
        for combo in permutations(numbers, r):
            items = [(v, str(v)) for v in combo]
            for val, expr in all_exprs(items):
                diff = abs(target - val)
                if diff < best_diff:
                    best_diff = diff
                    best = (val, expr)
                    if diff == 0:
                        return f"{expr} = {val}"
    
    if best:
        return f"{best[1]} = {best[0]}"
    return "No solution"

def main():
    data = sys.stdin.read().split('\n')
    target = int(data[0].strip())
    numbers = list(map(int, data[1].strip().split()))
    print(solve(target, numbers))

if __name__ == "__main__":
    main()
