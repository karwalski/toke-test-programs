import sys
from itertools import combinations

def parse_grid():
    lines = []
    for line in sys.stdin:
        lines.append(line.rstrip())
    while lines and not lines[-1]:
        lines.pop()
    grid = []
    for line in lines:
        grid.append(line.split())
    return grid

def solve_kakuro(grid):
    rows = len(grid)
    cols = len(grid[0])
    
    empty_cells = []
    h_clues = {}
    v_clues = {}
    
    for i in range(rows):
        for j in range(cols):
            cell = grid[i][j]
            if cell == '?':
                empty_cells.append((i, j))
            elif cell.startswith('H') and len(cell) > 1 and cell[1:].isdigit():
                h_clues[(i, j)] = int(cell[1:])
            elif cell.startswith('V') and len(cell) > 1 and cell[1:].isdigit():
                v_clues[(i, j)] = int(cell[1:])
    
    h_segments = {}
    v_segments = {}
    
    for (r, c), target_sum in h_clues.items():
        segment = []
        for j in range(c + 1, cols):
            if grid[r][j] == '?':
                segment.append((r, j))
            else:
                break
        if segment:
            h_segments[(r, c)] = (target_sum, segment)
    
    for (r, c), target_sum in v_clues.items():
        segment = []
        for i in range(r + 1, rows):
            if grid[i][c] == '?':
                segment.append((i, c))
            else:
                break
        if segment:
            v_segments[(r, c)] = (target_sum, segment)
    
    # Map cell -> list of segments it belongs to
    cell_segments = {}
    for key, (target, seg) in list(h_segments.items()) + list(v_segments.items()):
        for cell in seg:
            cell_segments.setdefault(cell, []).append((target, seg))
    
    assignment = {}
    
    def check_segment(target, seg):
        assigned = [assignment[c] for c in seg if c in assignment]
        if len(set(assigned)) != len(assigned):
            return False
        unassigned = len(seg) - len(assigned)
        cur = sum(assigned)
        if unassigned == 0:
            return cur == target
        remaining = target - cur
        if remaining <= 0:
            return False
        used = set(assigned)
        available = [d for d in range(1, 10) if d not in used]
        if len(available) < unassigned:
            return False
        min_p = sum(available[:unassigned])
        max_p = sum(available[-unassigned:])
        return min_p <= remaining <= max_p
    
    def backtrack(idx):
        if idx == len(empty_cells):
            for target, seg in h_segments.values():
                vals = [assignment[c] for c in seg]
                if sum(vals) != target or len(set(vals)) != len(vals):
                    return False
            for target, seg in v_segments.values():
                vals = [assignment[c] for c in seg]
                if sum(vals) != target or len(set(vals)) != len(vals):
                    return False
            return True
        
        cell = empty_cells[idx]
        segs = cell_segments.get(cell, [])
        
        for digit in range(1, 10):
            assignment[cell] = digit
            ok = True
            for target, seg in segs:
                if not check_segment(target, seg):
                    ok = False
                    break
            if ok and backtrack(idx + 1):
                return True
            del assignment[cell]
        return False
    
    if backtrack(0):
        out = [row[:] for row in grid]
        for (r, c), v in assignment.items():
            out[r][c] = str(v)
        # Strip H/V prefixes from clues
        for i in range(rows):
            for j in range(cols):
                cell = out[i][j]
                if cell.startswith('H') and len(cell) > 1 and cell[1:].isdigit():
                    out[i][j] = cell[1:]
                elif cell.startswith('V') and len(cell) > 1 and cell[1:].isdigit():
                    out[i][j] = cell[1:]
        return out
    return None

def main():
    grid = parse_grid()
    solution = solve_kakuro(grid)
    if solution:
        print('\n'.join(' '.join(row) for row in solution))
    else:
        print('No solution')

if __name__ == '__main__':
    main()
