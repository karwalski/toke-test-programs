```python
import sys

def main():
    data = sys.stdin.read().split('\n')
    grid_size = int(data[0].strip())
    cards = list(map(int, data[1].split()))
    
    matched = [False] * (grid_size * grid_size)
    score = 0
    flips = 0
    
    for line in data[2:]:
        line = line.strip()
        if not line:
            continue
        parts = line.split()
        if len(parts) < 2:
            continue
        pos1, pos2 = int(parts[0]), int(parts[1])
        flips += 1
        if (not matched[pos1] and not matched[pos2] and 
            pos1 != pos2 and cards[pos1] == cards[pos2]):
            matched[pos1] = True
            matched[pos2] = True
            score += 1
    
    remaining = flips - score
    print(f"Score: {score}")
    print(f"Remaining: {remaining}")

if __name__ == "__main__":
    main()
```
